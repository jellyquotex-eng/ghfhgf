/*
 * ESP32 Smart Safety System — FINAL v7 (Firebase)
 * - Buzzer : ultrasonic -> distance-based beep, water -> 2 Hz blink
 * - Motor  : CONTINUOUS ON while object in zone OR water detected
 * - Water  : auto-calibrates dry baseline at boot (wet = value DROPS)
 * - GPS    : auto-baud, last known location saved to flash
 * - Firebase: location pushed every 5 s from a SEPARATE task (core 0),
 *             so buzzer / motor / ultrasonic timing is never blocked.
 */

#include <WiFi.h>
#include <WiFiClientSecure.h>
#include <HTTPClient.h>
#include <Preferences.h>
#include <TinyGPSPlus.h>
#include <HardwareSerial.h>

// ---------------- WIFI ----------------
const char* WIFI_SSID = "TECNO CAMON 40 Pro";
const char* WIFI_PASS = "12345678";

// ---------------- FIREBASE ----------------
const char* FB_URL    = "https://my-website-ctg-default-rtdb.firebaseio.com";
const char* FB_SECRET = "";          // optional: Database secret. Khali thakle auth lage na
const uint32_t PUSH_PERIOD_MS = 5000;

// ---------------- PINS ----------------
#define PIN_TRIG        5
#define PIN_ECHO        18
#define PIN_BUZZER      14
#define PIN_MOTOR       26
#define PIN_WATER       34
#define GPS_RX_PIN      16   // GPS TX -> ESP32 GPIO16
#define GPS_TX_PIN      17

// ---------------- SETTINGS ----------------
#define ENABLE_WATER_ALERT   true
#define WATER_WET_IS_HIGHER  false   // tomar sensor-e panite value kome

const float    DETECT_ENTER_CM   = 50.0f;
const float    DETECT_RELEASE_CM = 55.0f;
const float    MAX_RANGE_CM      = 400.0f;
const uint32_t ULTRASONIC_PERIOD = 60;
const uint32_t WATER_PERIOD      = 200;
const unsigned long ECHO_TIMEOUT_US = 25000UL;
const uint8_t  NO_ECHO_CLEAR_COUNT  = 3;
const uint8_t  CONFIRM_HITS         = 2;

const int      WATER_DELTA         = 600;
const int      WATER_HYST          = 300;
const uint8_t  WATER_CONFIRM_COUNT = 5;
const uint32_t WATER_CAL_MS        = 1500;

const uint32_t BEEP_ON_MS = 100;

// ---------------- OBJECTS ----------------
TinyGPSPlus    gps;
HardwareSerial gpsSerial(2);
Preferences    prefs;

// ---------------- STATE ----------------
uint32_t lastUltrasonicMs = 0;
float    filteredDistance = MAX_RANGE_CM;
float    rawHistory[3]    = { MAX_RANGE_CM, MAX_RANGE_CM, MAX_RANGE_CM };
uint8_t  histIdx = 0, noEchoCount = 0, hitCount = 0;
bool     objectInZone = false;
uint32_t cycleStartMs = 0;

uint32_t lastWaterMs = 0;
int      waterRaw = 0, waterBaseline = 0;
bool     waterCalibrated = false, waterDetected = false;
uint8_t  waterWetCount = 0, waterDryCount = 0;

// GPS / location (shared with push task)
volatile double lastLat = 0, lastLng = 0;
volatile bool   hasLast = false;
volatile bool   gpsLive = false;
volatile uint32_t gpsSats = 0;
volatile float  gpsSpeed = 0;
double   savedLat = 0, savedLng = 0;
uint32_t lastSaveMs = 0, lastGpsPrintMs = 0, lastWifiTry = 0;

// =====================================================
//  WiFi
// =====================================================
void connectWiFi() {
  WiFi.mode(WIFI_STA);
  WiFi.setAutoReconnect(true);
  WiFi.begin(WIFI_SSID, WIFI_PASS);
  Serial.printf("[WIFI] Connecting to %s", WIFI_SSID);
  uint32_t t = millis();
  while (WiFi.status() != WL_CONNECTED && millis() - t < 15000) {
    delay(300);
    Serial.print(".");
  }
  Serial.println();
  if (WiFi.status() == WL_CONNECTED) {
    Serial.print("[WIFI] Connected. IP: ");
    Serial.println(WiFi.localIP());
  } else {
    Serial.println("[WIFI] Failed, retrying in background");
  }
}

void keepWiFiAlive() {
  if (WiFi.status() == WL_CONNECTED) return;
  if (millis() - lastWifiTry < 10000) return;
  lastWifiTry = millis();
  Serial.println("[WIFI] Reconnecting...");
  WiFi.disconnect();
  WiFi.begin(WIFI_SSID, WIFI_PASS);
}

// =====================================================
//  Firebase push task (runs on core 0, never blocks loop())
// =====================================================
void pushTask(void* arg) {
  for (;;) {
    vTaskDelay(PUSH_PERIOD_MS / portTICK_PERIOD_MS);
    if (!hasLast || WiFi.status() != WL_CONNECTED) continue;

    char body[220];
    snprintf(body, sizeof(body),
      "{\"lat\":%.6f,\"lng\":%.6f,\"sats\":%lu,\"speed\":%.2f,\"live\":%s,"
      "\"ts\":{\".sv\":\"timestamp\"}}",
      (double)lastLat, (double)lastLng, (unsigned long)gpsSats,
      (double)gpsSpeed, gpsLive ? "true" : "false");

    String url = String(FB_URL) + "/location.json";
    if (strlen(FB_SECRET) > 0) url += String("?auth=") + FB_SECRET;

    WiFiClientSecure client;
    client.setInsecure();
    HTTPClient http;
    http.setTimeout(6000);
    if (!http.begin(client, url)) continue;
    http.addHeader("Content-Type", "application/json");
    int code = http.PUT((uint8_t*)body, strlen(body));
    http.end();
    Serial.printf("[PUSH] firebase status: %d\n", code);
  }
}

// =====================================================
//  GPS
// =====================================================
bool listenForNMEA(uint32_t baud, uint32_t ms) {
  gpsSerial.end();
  gpsSerial.begin(baud, SERIAL_8N1, GPS_RX_PIN, GPS_TX_PIN);
  uint32_t t = millis();
  int dollars = 0;
  while (millis() - t < ms) {
    while (gpsSerial.available()) {
      if (gpsSerial.read() == '$') dollars++;
    }
    if (dollars >= 2) return true;
    delay(5);
  }
  return false;
}

void detectGpsBaud() {
  const uint32_t bauds[] = { 9600, 38400, 115200, 4800 };
  for (uint8_t i = 0; i < 4; i++) {
    Serial.printf("[GPS] Trying %lu baud... ", (unsigned long)bauds[i]);
    if (listenForNMEA(bauds[i], 1500)) {
      Serial.println("DATA OK");
      return;
    }
    Serial.println("no data");
  }
  Serial.println("[GPS] No data. Check wiring (GPS TX -> GPIO16, VCC, GND)");
  gpsSerial.end();
  gpsSerial.begin(9600, SERIAL_8N1, GPS_RX_PIN, GPS_TX_PIN);
}

void serviceGPS() {
  while (gpsSerial.available() > 0) {
    gps.encode(gpsSerial.read());
  }
  gpsSats = gps.satellites.isValid() ? gps.satellites.value() : 0;
  gpsSpeed = gps.speed.isValid() ? gps.speed.kmph() : 0;

  if (gps.location.isValid() && gps.location.age() < 5000 &&
      gps.location.lat() != 0.0) {
    lastLat = gps.location.lat();
    lastLng = gps.location.lng();
    hasLast = true;
    gpsLive = true;
  } else {
    gpsLive = false;
  }
}

void saveLocationIfNeeded() {
  if (!hasLast || !gpsLive) return;
  if (millis() - lastSaveMs < 30000) return;
  double la = lastLat, ln = lastLng;
  if (fabs(la - savedLat) < 0.0001 && fabs(ln - savedLng) < 0.0001 &&
      savedLat != 0) return;
  lastSaveMs = millis();
  savedLat = la; savedLng = ln;
  prefs.putDouble("lat", la);
  prefs.putDouble("lng", ln);
}

void printGPS() {
  Serial.println("---------- GPS ----------");
  if (gpsLive) {
    Serial.printf("Fix: VALID  Lat: %.6f  Lon: %.6f  Sat: %lu\n",
                  (double)lastLat, (double)lastLng, (unsigned long)gpsSats);
  } else {
    Serial.printf("Fix: NO FIX  Sat: %lu  Chars: %lu\n",
                  (unsigned long)gpsSats, (unsigned long)gps.charsProcessed());
    if (hasLast) Serial.printf("Last known: %.6f, %.6f\n", (double)lastLat, (double)lastLng);
  }
  Serial.printf("WiFi: %s\n", WiFi.status() == WL_CONNECTED ? "connected" : "disconnected");
  Serial.println("-------------------------");
}

// =====================================================
//  Ultrasonic
// =====================================================
float readUltrasonicCM() {
  digitalWrite(PIN_TRIG, LOW);
  delayMicroseconds(2);
  digitalWrite(PIN_TRIG, HIGH);
  delayMicroseconds(10);
  digitalWrite(PIN_TRIG, LOW);

  unsigned long dur = pulseIn(PIN_ECHO, HIGH, ECHO_TIMEOUT_US);
  if (dur == 0) return -1.0f;

  float cm = dur * 0.0343f / 2.0f;
  if (cm <= 2.0f || cm > MAX_RANGE_CM) return -1.0f;
  return cm;
}

float filterDistance(float newReading) {
  if (newReading < 0) {
    noEchoCount++;
    rawHistory[histIdx] = MAX_RANGE_CM;
    histIdx = (histIdx + 1) % 3;
    if (noEchoCount >= NO_ECHO_CLEAR_COUNT) filteredDistance = MAX_RANGE_CM;
    return filteredDistance;
  }
  noEchoCount = 0;

  rawHistory[histIdx] = newReading;
  histIdx = (histIdx + 1) % 3;

  float a = rawHistory[0], b = rawHistory[1], c = rawHistory[2];
  float med;
  if ((a <= b && b <= c) || (c <= b && b <= a))      med = b;
  else if ((b <= a && a <= c) || (c <= a && a <= b)) med = a;
  else                                               med = c;

  if (fabs(med - filteredDistance) > 30.0f) filteredDistance = med;
  else filteredDistance = 0.5f * med + 0.5f * filteredDistance;
  return filteredDistance;
}

uint32_t blinkIntervalForDistance(float cm) {
  if (cm <= 10.0f) return 200;
  if (cm <= 20.0f) return 300;
  if (cm <= 30.0f) return 500;
  if (cm <= 40.0f) return 700;
  return 1000;
}

// =====================================================
//  Outputs
// =====================================================
void updateOutputs() {
  uint32_t now = millis();
  bool proximityBeep = false;

  if (objectInZone) {
    uint32_t period = blinkIntervalForDistance(filteredDistance);
    uint32_t onMs   = (BEEP_ON_MS < period / 2) ? BEEP_ON_MS : period / 2;
    uint32_t phase  = now - cycleStartMs;
    if (phase >= period) { cycleStartMs = now; phase = 0; }
    proximityBeep = (phase < onMs);
  } else {
    cycleStartMs = now;
  }

  bool waterActive = ENABLE_WATER_ALERT && waterCalibrated && waterDetected;
  bool waterBeep   = waterActive && ((now / 250) & 1);

  digitalWrite(PIN_BUZZER, (proximityBeep || waterBeep) ? HIGH : LOW);
  digitalWrite(PIN_MOTOR,  (objectInZone || waterActive) ? HIGH : LOW);
}

// =====================================================
//  Water
// =====================================================
void calibrateWater() {
  Serial.println("[WATER] Calibrating dry baseline... keep sensor DRY");
  uint32_t start = millis();
  long sum = 0; int n = 0;
  while (millis() - start < WATER_CAL_MS) {
    sum += analogRead(PIN_WATER);
    n++;
    delay(20);
  }
  waterBaseline = (n > 0) ? (int)(sum / n) : 0;
  waterCalibrated = true;
  waterDetected = false;
  waterWetCount = waterDryCount = 0;
  Serial.printf("[WATER] Baseline = %d\n", waterBaseline);
}

void updateWater() {
  waterRaw = analogRead(PIN_WATER);
  if (!waterCalibrated) return;

  int change = WATER_WET_IS_HIGHER ? (waterRaw - waterBaseline)
                                   : (waterBaseline - waterRaw);

  if (change > WATER_DELTA) {
    waterDryCount = 0;
    if (waterWetCount < 255) waterWetCount++;
  } else if (change < (WATER_DELTA - WATER_HYST)) {
    waterWetCount = 0;
    if (waterDryCount < 255) waterDryCount++;
  }

  bool newState = waterDetected;
  if (!waterDetected && waterWetCount >= WATER_CONFIRM_COUNT) newState = true;
  if ( waterDetected && waterDryCount >= WATER_CONFIRM_COUNT) newState = false;

  if (newState != waterDetected) {
    waterDetected = newState;
    Serial.printf("[WATER] %s (raw=%d, baseline=%d)\n",
                  waterDetected ? "DETECTED" : "clear", waterRaw, waterBaseline);
  }
}

// =====================================================
//  SETUP
// =====================================================
void setup() {
  Serial.begin(115200);
  delay(200);
  Serial.println("\n=== ESP32 Smart Safety System booting ===");

  pinMode(PIN_TRIG, OUTPUT);
  pinMode(PIN_ECHO, INPUT);
  pinMode(PIN_BUZZER, OUTPUT);
  pinMode(PIN_MOTOR,  OUTPUT);
  digitalWrite(PIN_BUZZER, LOW);
  digitalWrite(PIN_MOTOR,  LOW);

  analogReadResolution(12);
  analogSetAttenuation(ADC_11db);

  prefs.begin("gps", false);
  savedLat = prefs.getDouble("lat", 0.0);
  savedLng = prefs.getDouble("lng", 0.0);
  if (savedLat != 0.0 || savedLng != 0.0) {
    lastLat = savedLat; lastLng = savedLng; hasLast = true;
    Serial.printf("[GPS] Loaded last known: %.6f, %.6f\n", savedLat, savedLng);
  }

  detectGpsBaud();
  calibrateWater();
  connectWiFi();

  // Firebase push alada task-e (core 0)
  xTaskCreatePinnedToCore(pushTask, "fbPush", 8192, NULL, 1, NULL, 0);

  Serial.println("Setup complete. Monitoring...\n");
}

// =====================================================
//  LOOP
// =====================================================
void loop() {
  uint32_t now = millis();

  serviceGPS();

  if (now - lastUltrasonicMs >= ULTRASONIC_PERIOD) {
    lastUltrasonicMs = now;
    float raw = readUltrasonicCM();
    filteredDistance = filterDistance(raw);

    if (!objectInZone) {
      if (raw > 0 && raw <= DETECT_ENTER_CM) {
        if (++hitCount >= CONFIRM_HITS) {
          objectInZone = true;
          cycleStartMs = millis();
          hitCount = 0;
        }
      } else {
        hitCount = 0;
      }
    } else if (filteredDistance > DETECT_RELEASE_CM) {
      objectInZone = false;
      hitCount = 0;
    }
  }

  if (now - lastWaterMs >= WATER_PERIOD) {
    lastWaterMs = now;
    updateWater();
  }

  updateOutputs();
  saveLocationIfNeeded();
  keepWiFiAlive();

  if (now - lastGpsPrintMs >= 2000) {
    lastGpsPrintMs = now;
    printGPS();
  }
}
